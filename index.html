               <div style="font-size:0.75rem; color:var(--text-muted); margin-top:8px; text-align:right;">20 Soalan Penuh</div>
           `;
           grid.appendChild(card);
       }
   };

   function switchScreen(screenId) {
       document.querySelectorAll(".screen").forEach(s => s.classList.remove("active"));
       document.getElementById(screenId).classList.add("active");
   }

   function startUnitQuiz(unitId) {
       activeUnit = unitId;
       currentQuestionIndex = 0;
       userScore = 0;

        document.getElementById("current-unit-title").innerText = `Unit ${unitId}: ${unitTitles[unitId]}`;
       switchScreen("quiz-screen");
       loadQuestion();
   }

   function loadQuestion() {
       questionAnswered = false;
       const qData = rawQuestions[activeUnit][currentQuestionIndex];
       
       document.getElementById("question-counter").innerText = `Soalan ${currentQuestionIndex + 1} / 20`;
       document.getElementById("quizProgressBar").style.width = `${(currentQuestionIndex / 20) * 100}%`;
       document.getElementById("questionText").innerText = qData.q;
       document.getElementById("explanationBox").style.display = "none";
       
       document.getElementById("nextBtn").disabled = true;
       document.getElementById("skipBtn").style.display = "inline-flex";

       const container = document.getElementById("optionsContainer");
       container.innerHTML = "";

       
       qData.a.forEach((opt, idx) => {
           const li = document.createElement("li");
           li.className = "option-item";
           li.innerText = opt;
           li.onclick = function() { checkAnswer(li, idx); };
           container.appendChild(li);
       });
   }

   function checkAnswer(element, selectedIdx) {
       if (questionAnswered) return;
       questionAnswered = true;
       
       const qData = rawQuestions[activeUnit][currentQuestionIndex];
       const items = document.querySelectorAll(".option-item");
       
       items.forEach(item => item.classList.add("disabled"));
       

        if (selectedIdx === qData.c) {
           userScore++;
           element.classList.add("selected-correct");
       } else {
           element.classList.add("selected-wrong");
           items[qData.c].classList.add("show-correct-target");
       }
       
       document.getElementById("explanationText").innerText = qData.e;
       document.getElementById("explanationBox").style.display = "block";
       
       document.getElementById("nextBtn").disabled = false;
       document.getElementById("skipBtn").style.display = "none";
   }

   function navigateQuestion(step) {
       currentQuestionIndex += step;
       if (currentQuestionIndex < 20) {
           loadQuestion();

        } else {
           showResults();
       }
   }

   function showResults() {
       document.getElementById("quizProgressBar").style.width = "100%";
       document.getElementById("finalScore").innerText = `${userScore}/20`;
       
       const status = document.getElementById("resultStatus");
       const comment = document.getElementById("resultComment");
       const pct = (userScore / 20) * 100;
       
       if (pct === 100) {
           status.innerText = "Cemerlang! ✨";
           comment.innerText = "Sempurna! Anda menguasai semua huraian fakta unit ini dengan sifar ralat.";
       } else if (pct >= 50) {
           status.innerText = "Tahniah, Anda Lulus! 👍";
           comment.innerText = "Gred yang baik. Sila ulang uji untuk menajamkan lagi ingatan jawapan akademik anda.";

        } else {
           status.innerText = "Sila Cuba Lagi 📚";
           comment.innerText = "Banyakkan membaca semula ulasan fakta kuiz untuk membetulkan kekeliruan sejarah.";
       }
       switchScreen("result-screen");
   }
